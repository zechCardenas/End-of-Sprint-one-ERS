from flask import Blueprint, render_template, request, redirect, url_for, flash
from flask_login import login_user, logout_user, login_required
from werkzeug.security import check_password_hash
from models import User  # your User model
from extensions import db, login_manager  # your initialized Flask extensions

auth = Blueprint('auth', __name__)

# -------------------------------------------------------
# LOGIN ROUTE
# -------------------------------------------------------
@auth.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        email = request.form.get('email')
        password = request.form.get('password')
        remember = True if request.form.get('remember') else False  # ✅ Remember Me

        # Find the user in DB
        user = User.query.filter_by(email=email).first()

        if not user or not check_password_hash(user.password, password):
            flash('Invalid email or password', 'error')
            return redirect(url_for('auth.login'))

        # Log in and remember session
        login_user(user, remember=remember)
        flash('Login successful!', 'success')
        return redirect(url_for('main.dashboard'))

    return render_template('login.html')


# -------------------------------------------------------
# LOGOUT ROUTE
# -------------------------------------------------------
@auth.route('/logout')
@login_required
def logout():
    logout_user()
    flash('You have been logged out.', 'info')
    return redirect(url_for('auth.login'))
